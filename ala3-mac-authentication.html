02 Sender Script — mac_sender.py
Python 3.x — Sender 
# ============================================================
# ALA 3 – MAC Sender Script
# Course: Information Security | GMIU
# ============================================================

import hmac
import hashlib
import base64
import json


# ── Shared Secret Key (must match receiver) ─────────────────
SECRET_KEY = b"GMIU_SecretKey_2024_InfoSec"


def generate_mac(message: str, key: bytes) -> str:
    """Generate HMAC-SHA256 MAC for a message using shared key."""
    mac = hmac.new(
        key,
        message.encode('utf-8'),
        hashlib.sha256
    )
    return base64.b64encode(mac.digest()).decode('utf-8')


def create_authenticated_packet(message: str) -> dict:
    """Create a packet with message and its MAC tag."""
    mac_tag = generate_mac(message, SECRET_KEY)
    return {
        "message": message,
        "mac_tag": mac_tag,
        "algorithm": "HMAC-SHA256"
    }


if __name__ == "__main__":
    print("=" * 60)
    print("  ALA 3 — MAC Sender")
    print("=" * 60)

    message = "Transfer Rs. 5000 to Account No. 987654321"
    packet = create_authenticated_packet(message)

    print(f"\n[SENDER] Original Message : {packet['message']}")
    print(f"[SENDER] MAC Tag (b64)    : {packet['mac_tag']}")
    print(f"[SENDER] Algorithm        : {packet['algorithm']}")

    # Simulate sending over network (save to file)
    with open("packet.json", "w") as f:
        json.dump(packet, f, indent=2)

    print("\n[SENDER] Packet transmitted → packet.json")
    print("=" * 60)
03 Receiver Script — mac_receiver.py
Python 3.x — Receiver 
# ============================================================
# ALA 3 – MAC Receiver Script
# Course: Information Security | GMIU
# ============================================================

import hmac
import hashlib
import base64
import json


# ── Same Shared Secret Key as sender ───────────────────────
SECRET_KEY = b"GMIU_SecretKey_2024_InfoSec"


def verify_mac(message: str, received_mac: str, key: bytes) -> bool:
    """
    Recompute MAC and compare using hmac.compare_digest
    to prevent timing attacks.
    """
    expected_mac = hmac.new(
        key,
        message.encode('utf-8'),
        hashlib.sha256
    )
    expected_b64 = base64.b64encode(expected_mac.digest()).decode('utf-8')
    # constant-time comparison prevents timing side-channel attacks
    return hmac.compare_digest(expected_b64, received_mac)


if __name__ == "__main__":
    print("=" * 60)
    print("  ALA 3 — MAC Receiver")
    print("=" * 60)

    # Load received packet
    with open("packet.json", "r") as f:
        packet = json.load(f)

    print(f"\n[RECEIVER] Received Message : {packet['message']}")
    print(f"[RECEIVER] Received MAC     : {packet['mac_tag']}")

    # Case 1: Verify original (unmodified) message
    result = verify_mac(packet['message'], packet['mac_tag'], SECRET_KEY)
    status = "✅ VERIFIED — Message is authentic!" if result else "❌ FAILED!"
    print(f"\n[RECEIVER] Verification      : {status}")

    # Case 2: Simulate tampering — attacker modifies message
    tampered = packet['message'].replace("5000", "50000")
    tampered_result = verify_mac(tampered, packet['mac_tag'], SECRET_KEY)
    t_status = "✅ VERIFIED" if tampered_result else "❌ TAMPERED — MAC mismatch!"

    print(f"\n[RECEIVER] Tampered Message  : {tampered}")
    print(f"[RECEIVER] Tampered Check    : {t_status}")
    print("\n[!] HMAC successfully detected the tampering!")
    print("=" * 60)

05 Analysis & Key Concepts
🔐

Shared Secret

HMAC uses a symmetric shared secret key known only to sender and receiver. Anyone without the key cannot forge a valid MAC.

🛡️

Data Integrity

Even a single character change in the message produces a completely different MAC, making any tampering immediately detectable.

👤

Authentication

Because only parties with the shared key can compute the correct MAC, the receiver can confirm the message's origin.

⏱️

Timing Attack Safe

Using hmac.compare_digest() instead of == prevents timing side-channel attacks where an attacker could brute-force the MAC byte by byte.

⚔️

MAC vs Signature

MAC requires a shared key (symmetric) — both parties can generate it. Digital Signatures use asymmetric keys — only the signer can sign.

🌐

Real-World Use

HMAC is used in JWT tokens, TLS 1.3, API authentication (AWS Signature), WhatsApp message authentication, and VPN tunnels.
