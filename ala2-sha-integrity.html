import hashlib


# ── Hash a message with multiple SHA algorithms ─────────────
def compute_hashes(message: str) -> dict:
    """Return SHA-1, SHA-256, SHA-512 hashes of a message."""
    encoded = message.encode('utf-8')
    return {
        'SHA-1'  : hashlib.sha1(encoded).hexdigest(),
        'SHA-256': hashlib.sha256(encoded).hexdigest(),
        'SHA-512': hashlib.sha512(encoded).hexdigest(),
    }


# ── Compare two hash sets and count differing bits ──────────
def bit_difference(h1: str, h2: str) -> float:
    """Calculate percentage of bits that differ (avalanche)."""
    b1 = bin(int(h1, 16))[2:].zfill(len(h1) * 4)
    b2 = bin(int(h2, 16))[2:].zfill(len(h2) * 4)
    diff = sum(c1 != c2 for c1, c2 in zip(b1, b2))
    return (diff / len(b1)) * 100


# ── Print formatted hash output ─────────────────────────────
def print_hashes(label: str, hashes: dict):
    print(f"\n  [{label}]")
    for algo, digest in hashes.items():
        print(f"  {algo:<8}: {digest}")


# ── Main Program ───────────────────────────────────────────
if __name__ == "__main__":
    print("=" * 75)
    print("  ALA 2 — SHA Integrity Analyzer  |  Avalanche Effect Demo")
    print("=" * 75)

    # Original and slightly changed messages
    msg1 = "The quick brown fox jumps over the lazy dog"
    msg2 = "The quick brown fox jumps over the lazy Dog"  # 'd' → 'D'

    h1 = compute_hashes(msg1)
    h2 = compute_hashes(msg2)

    print_hashes("Original Message", h1)
    print_hashes("Modified Message (1 char change: d→D)", h2)

    # Show avalanche effect
    print("\n" + "─" * 75)
    print("  AVALANCHE EFFECT — Bit Difference After 1 Character Change:")
    print("─" * 75)
    for algo in ['SHA-1', 'SHA-256', 'SHA-512']:
        diff = bit_difference(h1[algo], h2[algo])
        print(f"  {algo:<8}: {diff:.1f}% of bits changed")

    # Fixed-length test
    print("\n" + "─" * 75)
    print("  FIXED LENGTH TEST — Any input gives same digest size:")
    short_h = compute_hashes("a")
    long_h  = compute_hashes("a" * 1000000)
    for algo in ['SHA-1', 'SHA-256', 'SHA-512']:
        print(f"  {algo:<8}  'a'={len(short_h[algo])} chars, 'a'x1M={len(long_h[algo])} chars")

    print("=" * 75)

05 Analysis & Key Observations
🌊

Avalanche Effect

~50% of bits change with even a 1-character input change. This ensures hashes are unpredictable and collision-resistant.

📏

Fixed Output Size

SHA-256 always outputs exactly 64 hex characters, regardless of whether the input is 1 byte or 1 GB in size.

🚫

One-Way Function

Hash functions are irreversible — you cannot derive the original input from the hash. This is fundamental to security.

⚠️

SHA-1 Deprecated

SHA-1 was broken by Google's SHAttered attack in 2017 using a chosen-prefix collision. Never use it for security.

🔐

Digital Signatures

Hash functions allow signing large documents efficiently — you sign the hash (fixed size), not the entire document.

🛡️

Integrity Check

SHA-256 is used in TLS certificates, file integrity verification, and blockchain to detect any data modification.
