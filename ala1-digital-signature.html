from cryptography.hazmat.primitives.asymmetric import rsa, padding
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.exceptions import InvalidSignature
import base64


# ── Step 1: Generate RSA Key Pair ──────────────────────────
def generate_key_pair():
    """Generate a 2048-bit RSA private/public key pair."""
    private_key = rsa.generate_private_key(
        public_exponent=65537,
        key_size=2048
    )
    public_key = private_key.public_key()
    return private_key, public_key


# ── Step 2: Sign a Message ─────────────────────────────────
def sign_message(private_key, message: str) -> bytes:
    """Sign a message using RSA-PSS with SHA-256 hashing."""
    signature = private_key.sign(
        message.encode('utf-8'),
        padding.PSS(
            mgf=padding.MGF1(hashes.SHA256()),
            salt_length=padding.PSS.MAX_LENGTH
        ),
        hashes.SHA256()
    )
    return signature


# ── Step 3: Verify the Signature ───────────────────────────
def verify_signature(public_key, message: str, signature: bytes) -> bool:
    """Verify a digital signature using the public key."""
    try:
        public_key.verify(
            signature,
            message.encode('utf-8'),
            padding.PSS(
                mgf=padding.MGF1(hashes.SHA256()),
                salt_length=padding.PSS.MAX_LENGTH
            ),
            hashes.SHA256()
        )
        return True
    except InvalidSignature:
        return False


# ── Step 4: Export Keys (PEM Format) ───────────────────────
def export_keys(private_key, public_key):
    priv_pem = private_key.private_bytes(
        encoding=serialization.Encoding.PEM,
        format=serialization.PrivateFormat.TraditionalOpenSSL,
        encryption_algorithm=serialization.NoEncryption()
    )
    pub_pem = public_key.public_bytes(
        encoding=serialization.Encoding.PEM,
        format=serialization.PublicFormat.SubjectPublicKeyInfo
    )
    return priv_pem.decode(), pub_pem.decode()


# ── Main Program ───────────────────────────────────────────
if __name__ == "__main__":
    print("=" * 55)
    print("  ALA 1 — Digital Signature Verifier")
    print("=" * 55)

    # Generate keys
    private_key, public_key = generate_key_pair()
    priv_pem, pub_pem = export_keys(private_key, public_key)
    print("\n[+] RSA 2048-bit Key Pair Generated Successfully")
    print("\nPublic Key (PEM):\n", pub_pem[:90], "...")

    # Sign message
    message = "Hello, this message is signed by GMIU Student!"
    signature = sign_message(private_key, message)
    sig_b64 = base64.b64encode(signature).decode()

    print(f"\n[+] Message : {message}")
    print(f"[+] Signature (b64): {sig_b64[:64]}...")

    # Verify valid signature
    result = verify_signature(public_key, message, signature)
    print(f"\n[✓] Signature Valid  : {result}")

    # Tamper with message and verify again
    tampered = message + " [TAMPERED]"
    tampered_result = verify_signature(public_key, tampered, signature)
    print(f"[✗] Tampered Msg Valid: {tampered_result}")
    print("\n[!] Data Integrity Confirmed — Tampering Detected!")
    print("=" * 55)

04 Analysis & Key Concepts
🔑

Asymmetric Keys

RSA uses a mathematically linked key pair. The private key signs; the public key verifies. The private key cannot be derived from the public key.

🔐

SHA-256 Hashing

The message is first hashed with SHA-256 to produce a fixed-length digest, then the digest is signed — not the full message — for efficiency and security.

✅

Non-Repudiation

Only the holder of the private key could have produced the signature. This property proves authorship and prevents the signer from denying the signed message.

🛡️

Integrity

Any change to the message — even a single character — causes signature verification to fail, immediately revealing tampering.

📋

PSS Padding

Probabilistic Signature Scheme (PSS) adds randomness to each signature, making it resistant to chosen-message attacks unlike deterministic PKCS#1 v1.5.

🌐

Real-World Use

Digital signatures are used in SSL/TLS certificates, code signing, email (PGP/S-MIME), PDF signing, and blockchain transaction verification.
